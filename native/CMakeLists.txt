cmake_minimum_required(VERSION 3.20)
project(cpp_hint_ast LANGUAGES C CXX)

find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
add_executable(cpp-hint-ast main.cpp)
target_compile_features(cpp-hint-ast PRIVATE cxx_std_17)
target_include_directories(cpp-hint-ast SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
if(MSVC)
  target_compile_options(cpp-hint-ast PRIVATE /bigobj /utf-8)
endif()
if(TARGET clang-cpp AND NOT MSVC)
  target_link_libraries(cpp-hint-ast PRIVATE clang-cpp)
else()
  target_link_libraries(cpp-hint-ast PRIVATE clangTooling clangFrontend clangAST clangLex clangBasic)
endif()

install(TARGETS cpp-hint-ast RUNTIME DESTINATION .)
install(FILES LICENSE-LLVM.txt LICENSE-zlib.txt LICENSE-zstd.txt DESTINATION .)
set(CLANG_RESOURCE_DIR "" CACHE PATH "Directory containing Clang's builtin include/ headers")
if(CLANG_RESOURCE_DIR)
  install(DIRECTORY "${CLANG_RESOURCE_DIR}/include" DESTINATION resource)
endif()
